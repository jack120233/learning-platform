# 系统管理模块技术详情文档

## 模块概述

系统管理模块提供平台基础配置管理功能，包括**分类管理**、**标签管理**和**公告管理**三大子模块。分类和标签用于课程分类与检索，公告用于平台运营信息发布。本模块支持多级分类结构，为课程管理和搜索功能提供基础数据支撑。

#### 5. 数据库操作

**涉及数据表**：`categories`

**表结构**：
```sql
CREATE TABLE categories (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(50) UNIQUE NOT NULL COMMENT '分类名称',
    parent_id BIGINT DEFAULT NULL COMMENT '父分类ID',
    icon_url VARCHAR(255) DEFAULT NULL COMMENT '分类图标URL',
    sort_order INT DEFAULT 0 COMMENT '排序序号',
    is_enabled BOOLEAN DEFAULT TRUE COMMENT '是否启用',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_parent_id (parent_id),
    INDEX idx_is_enabled (is_enabled),
    FOREIGN KEY (parent_id) REFERENCES categories(id) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
```

**ORM查询示例**：
```python
async def get_categories(
    db: AsyncSession,
    parent_id: Optional[int] = None,
    is_enabled: Optional[bool] = True
) -> List[Category]:
    """获取分类列表"""
    query = select(Category)

    # 筛选条件
    if parent_id is not None:
        query = query.where(Category.parent_id == parent_id)
    else:
        query = query.where(Category.parent_id.is_(None))

    if is_enabled is not None:
        query = query.where(Category.is_enabled == is_enabled)

    # 排序
    query = query.order_by(Category.sort_order.asc())

    result = await db.execute(query)
    return result.scalars().all()
```

**递归加载子分类**：
```python
async def build_category_tree(
    db: AsyncSession,
    categories: List[Category],
    is_enabled: Optional[bool] = True
) -> List[dict]:
    """构建分类树"""
    tree = []
    for category in categories:
        node = CategoryResponse.model_validate(category)

        # 递归加载子分类
        children = await get_categories(db, parent_id=category.id, is_enabled=is_enabled)
        if children:
            node.children = await build_category_tree(db, children, is_enabled)

        tree.append(node)
    return tree
```

#### 2. 请求参数

┌─────────────────────────────────────────────────────────────┐
│                      创建分类流程                              │

│     - 检查父分类是否存在                                       │
│     - 检查分类层级（最多支持2级）                               │

**关键决策点**：
- 分类层级限制：最多支持2级（一级分类 → 二级分类）
- 名称唯一性：全局唯一，不区分一级/二级分类
- 排序自动计算：如果未传`sort_order`，自动设置为同级分类最大值+1

async def get_announcements(
    db: AsyncSession,
    status: Optional[str] = "published",
    page: int = 1,
    page_size: int = 20
) -> Tuple[List[Announcement], int]:
    """获取公告列表"""
    # 构建查询
    query = select(Announcement).options(
        joinedload(Announcement.creator)
    )

    # 状态筛选
    if status:
        query = query.where(Announcement.status == status)

    # 排序（发布时间倒序）
    query = query.order_by(Announcement.published_at.desc().nulls_last())

    # 分页
    count_query = select(func.count()).select_from(query.subquery())
    total = (await db.execute(count_query)).scalar()

    query = query.offset((page - 1) * page_size).limit(page_size)
    result = await db.execute(query)
    items = result.scalars().unique().all()

    return items, total

## 五、服务层实现示例

### 5.1 分类服务

```python
# services/category_service.py
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select, func
from typing import List, Optional
from app.models.category import Category
from app.schemas.category import CategoryCreate, CategoryUpdate
from app.core.exceptions import NotFoundException, ConflictException, BusinessException
from app.services.cache_service import CacheService

class CategoryService:
    """分类服务"""

    CACHE_KEY_TREE = "categories:tree"
    CACHE_TTL = 3600  # 1小时

    def __init__(self, db: AsyncSession, cache: CacheService):
        self.db = db
        self.cache = cache

    async def get_tree(
        self,
        parent_id: Optional[int] = None,
        is_enabled: Optional[bool] = True
    ) -> List[dict]:
        """获取分类树"""
        # 尝试从缓存获取
        if parent_id is None and is_enabled:
            cached = await self.cache.get(self.CACHE_KEY_TREE)
            if cached:
                return cached

        # 查询数据库
        categories = await self._get_by_parent(parent_id, is_enabled)
        tree = await self._build_tree(categories, is_enabled)

        # 缓存结果
        if parent_id is None and is_enabled:
            await self.cache.set(self.CACHE_KEY_TREE, tree, self.CACHE_TTL)

        return tree

    async def _get_by_parent(
        self,
        parent_id: Optional[int],
        is_enabled: Optional[bool]
    ) -> List[Category]:
        """获取指定父分类下的子分类"""
        query = select(Category)

        if parent_id is not None:
            query = query.where(Category.parent_id == parent_id)
        else:
            query = query.where(Category.parent_id.is_(None))

        if is_enabled is not None:
            query = query.where(Category.is_enabled == is_enabled)

        query = query.order_by(Category.sort_order.asc())
        result = await self.db.execute(query)
        return list(result.scalars().all())

    async def _build_tree(
        self,
        categories: List[Category],
        is_enabled: Optional[bool]
    ) -> List[dict]:
        """递归构建分类树"""
        tree = []
        for category in categories:
            node = {
                "category_id": category.id,
                "name": category.name,
                "parent_id": category.parent_id,
                "icon_url": category.icon_url,
                "sort_order": category.sort_order,
                "is_enabled": category.is_enabled,
                "children": []
            }

            # 递归加载子分类
            children = await self._get_by_parent(category.id, is_enabled)
            if children:
                node["children"] = await self._build_tree(children, is_enabled)

            tree.append(node)
        return tree

    async def create(self, data: CategoryCreate) -> Category:
        """创建分类"""
        # 检查名称唯一性
        if await self._name_exists(data.name):
            raise ConflictException("分类名称已存在")

        # 检查父分类
        if data.parent_id:
            parent = await self.db.get(Category, data.parent_id)
            if not parent:
                raise NotFoundException("父分类不存在")

            # 检查层级（最多2级）
            if parent.parent_id is not None:
                raise BusinessException("最多支持2级分类")

        # 创建分类
        category = Category(
            name=data.name,
            parent_id=data.parent_id,
            icon_url=data.icon_url,
            sort_order=data.sort_order or 0,
            is_enabled=True
        )
        self.db.add(category)
        await self.db.commit()
        await self.db.refresh(category)

        # 清除缓存
        await self._invalidate_cache()

        return category

    async def update(self, category_id: int, data: CategoryUpdate) -> Category:
        """更新分类"""
        category = await self.db.get(Category, category_id)
        if not category:
            raise NotFoundException("分类不存在")

        # 检查名称唯一性
        if data.name and data.name != category.name:
            if await self._name_exists(data.name, exclude_id=category_id):
                raise ConflictException("分类名称已存在")

        # 更新字段
        update_data = data.model_dump(exclude_unset=True)
        for field, value in update_data.items():
            setattr(category, field, value)

        await self.db.commit()
        await self.db.refresh(category)

        # 清除缓存
        await self._invalidate_cache()

        return category

    async def delete(self, category_id: int) -> None:
        """删除分类"""
        category = await self.db.get(Category, category_id)
        if not category:
            raise NotFoundException("分类不存在")

        # 检查课程关联
        if await self._has_courses(category_id):
            raise BusinessException(
                "该分类已关联课程，无法删除，请先移除关联或禁用分类"
            )

        await self.db.delete(category)
        await self.db.commit()

        # 清除缓存
        await self._invalidate_cache()

    async def _name_exists(self, name: str, exclude_id: int = None) -> bool:
        """检查名称是否存在"""
        query = select(Category).where(Category.name == name)
        if exclude_id:
            query = query.where(Category.id != exclude_id)
        result = await self.db.execute(query)
        return result.scalar_one_or_none() is not None

    async def _has_courses(self, category_id: int) -> bool:
        """检查分类是否关联课程"""
        from app.models.course import Course
        query = select(func.count()).select_from(Course).where(
            Course.category_id == category_id
        )
        result = await self.db.execute(query)
        return result.scalar() > 0

    async def _invalidate_cache(self):
        """清除分类缓存"""
        await self.cache.delete(self.CACHE_KEY_TREE)
```

## 七、注意事项

1. **分类层级限制**：最多支持2级分类结构（一级分类 → 二级分类）
2. **分类删除限制**：已关联课程的分类不可删除，只能禁用
3. **名称唯一性**：分类名称和标签名称全局唯一
4. **缓存策略**：分类列表和标签列表建议使用Redis缓存，变化频率低
5. **权限控制**：创建、更新、删除操作需要讲师或管理员权限
6. **公告可见性**：普通用户只能查看已发布公告，管理员可查看所有状态
