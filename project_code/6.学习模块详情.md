## 模块概述

学习模块负责处理学员的学习行为，包括开始学习课程、保存学习进度、获取学习进度、继续学习以及获取资源播放地址等功能。本模块是学员与课程内容交互的核心模块，共包含6个接口。

## 一、开始学习课程

### 1. 基本信息

| 项目 | 内容 |
|------|------|
| 接口名称 | 开始学习课程 |
| 接口路径 | /api/v1/learning/courses/{course_id}/start |
| HTTP方法 | POST |
| 功能描述 | 学员首次学习某课程时创建学习记录，记录首次学习时间 |

### 2. 请求参数

#### Path参数

| 参数 | 类型 | 必填 | 说明 |
|------|------|------|------|
| course_id | integer | 是 | 课程ID |

### 3. 响应结构

#### 成功响应

```json
{
  "code": 200,
  "message": "开始学习",
  "data": {
    "course_id": 1,
    "last_section_id": null,
    "created_at": "2026-03-17T10:00:00"
  }
}
```

#### 响应字段说明

| 字段 | 类型 | 说明 |
|------|------|------|
| course_id | integer | 课程ID |
| last_section_id | integer | 最后学习的小节ID（首次为null） |
| created_at | datetime | 首次学习时间 |

#### 失败响应

| 场景 | code | message |
|------|------|---------|
| 未登录 | 401 | 未认证或令牌过期 |
| 课程不存在 | 404 | 课程不存在 |
| 课程未发布 | 400 | 课程暂未发布，无法学习 |
| 已有学习记录 | 200 | 已存在学习记录（幂等返回） |

### 4. 业务逻辑

```
开始
  ↓
验证用户登录状态
  ├── 未登录 → 返回401错误
  └── 已登录 → 继续
  ↓
验证课程存在性
  ├── 不存在 → 返回404错误
  └── 存在 → 继续
  ↓
验证课程状态
  ├── 未发布 → 返回400错误
  └── 已发布 → 继续
  ↓
检查是否已有学习记录
  ├── 有 → 返回已有记录（幂等）
  └── 无 → 创建学习记录
  ↓
创建课程学习记录
  ├── user_id = current_user.id
  ├── course_id = course_id
  ├── first_learn_at = now
  └── last_learn_at = now
  ↓
返回学习记录信息
结束
```

- **涉及数据表**：learning_progress
- **操作类型**：查询、插入

from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.ext.asyncio import AsyncSession

router = APIRouter(prefix="/learning", tags=["学习模块"])

@router.post("/courses/{course_id}/start", response_model=ApiResponse[StartLearningResponse])
async def start_learning(
    course_id: int,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db)
):
    # 验证课程
    course = await course_service.get_course_by_id(db, course_id)

        raise HTTPException(status_code=404, detail="课程不存在")
    if course.status != 'published':
        raise HTTPException(status_code=400, detail="课程暂未发布，无法学习")

    # 开始学习
    progress = await learning_service.start_learning(db, current_user.id, course_id)

    return ApiResponse(
        message="开始学习",
        data=StartLearningResponse(
            course_id=course_id,
            last_section_id=progress.last_section_id,
            created_at=progress.first_learn_at
        )
    )

# SQLAlchemy示例
async with db.begin():
    # 获取课程ID
    section = await db.get(Section, section_id)
    chapter = await db.get(Chapter, section.chapter_id)
    course_id = chapter.course_id

    # 检查课程学习记录

    resource = await db.get(Resource, resource_id)
    if resource.resource_type in ['video', 'audio']:
        is_completed = current_time >= total_time * 0.95
    else:
        is_completed = True

    # 保存资源进度（upsert）
    existing = await db.execute(
        select(ResourceProgress)
        .where(
            ResourceProgress.user_id == user_id,
            ResourceProgress.resource_id == resource_id

    if existing.scalar():

            update(ResourceProgress)
            .where(ResourceProgress.id == existing.scalar().id)

                last_learn_at=datetime.now()
            )

            user_id=user_id,
            resource_id=resource_id,
            section_id=section_id,
            current_time=current_time,
            total_time=total_time,
            is_completed=is_completed

    # 更新课程学习记录

        update(LearningProgress)
        .where(
            LearningProgress.user_id == user_id,
            LearningProgress.course_id == course_id

### 10. 代码组织建议

#### Pydantic模型

```python
# schemas/learning.py
from pydantic import BaseModel, Field, field_validator

class SaveProgressRequest(BaseModel):
    section_id: int = Field(..., gt=0)
    resource_id: int = Field(..., gt=0)
    current_time: int = Field(..., ge=0)
    total_time: int = Field(..., gt=0)
    is_completed: bool = False

    @field_validator('current_time')
    @classmethod
    def validate_current_time(cls, v, info):
        if 'total_time' in info.data and v > info.data['total_time']:
            raise ValueError('当前时间不能大于总时长')
        return v
```

#### 服务层

```python
# services/learning_service.py
class LearningService:
    async def save_progress(
        self,
        db: AsyncSession,
        user_id: int,
        request: SaveProgressRequest
    ) -> None:
        """保存学习进度"""
        # 验证小节和资源
        section = await db.get(Section, request.section_id)
        if not section:
            raise ValueError("小节不存在")

        resource = await db.get(Resource, request.resource_id)
        if not resource:
            raise ValueError("资源不存在")

        if resource.section_id != request.section_id:
            raise ValueError("资源与小节不匹配")

        # 获取课程ID
        chapter = await db.get(Chapter, section.chapter_id)
        course_id = chapter.course_id

        # 检查课程学习记录
        course_progress = await self.get_course_progress(db, user_id, course_id)
        if not course_progress:
            raise ValueError("请先开始学习课程")

        # 计算完成状态
        is_completed = self._calculate_completion(
            resource.resource_type,
            request.current_time,
            request.total_time
        )

        # 保存进度
        await self._save_resource_progress(
            db, user_id, request, is_completed
        )

        # 更新课程学习记录
        await self._update_course_progress(
            db, user_id, course_id, request.section_id, request.resource_id
        )

    def _calculate_completion(
        self,
        resource_type: str,
        current_time: int,
        total_time: int
    ) -> bool:
        """计算完成状态"""
        if resource_type in ['video', 'audio']:
            # 视频和音频播放到95%以上视为完成
            return current_time >= total_time * 0.95
        else:
            # 图片和文档打开即完成
            return True
```

# api/v1/learning.py
@router.get("/resources/{resource_id}/play", response_model=ApiResponse[ResourcePlayResponse])
async def get_resource_play_url(
    resource_id: int,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db)
):
    # 获取资源信息

            duration=resource.duration,
            resolution=resource.resolution,
            thumbnail_url=resource.thumbnail_url

from sqlalchemy import Column, Integer, DateTime, ForeignKey

    first_learn_at = Column(DateTime, default=datetime.utcnow)
    last_learn_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)
    last_section_id = Column(Integer, ForeignKey("sections.id"), nullable=True)
    last_resource_id = Column(Integer, ForeignKey("resources.id"), nullable=True)

    # 关系
    user = relationship("User", back_populates="learning_progress")
    course = relationship("Course", back_populates="learning_progress")
    last_section = relationship("Section")
    last_resource = relationship("Resource")

    __table_args__ = (
        UniqueConstraint('user_id', 'course_id', name='uq_user_course'),
    )
