## 模块概述

用户管理模块负责处理用户个人信息管理、学习记录查询、用户名修改机会管理以及管理员对用户的各项操作（用户列表、禁用/启用、删除、审核等）。本模块当前实现包含用户资料、开放改名机会、老师选择项、我的反馈、用户列表、审核等接口，覆盖普通用户、讲师和管理员角色的操作场景。

### 当前实现补充：用户名修改次数

- **数据库表**：`users`
- **剩余改名次数字段**：`username_change_remaining`，类型 `Integer`，默认值 `1`，含义为“剩余用户名修改次数”。
- **原用户名字段**：`original_username`，类型 `String(50)`，首次自助改名前记录原用户名。
- **接口派生字段**：`can_change_username`，由后端响应模型根据角色和剩余次数计算，不是数据库字段。
- **规则**：`student` 等普通受限用户修改用户名会消耗 `username_change_remaining`；`teacher` 和 `admin` 不受次数限制，`can_change_username` 恒为 `true`，修改用户名时不扣减该字段。
- **兼容旧库**：`backend/app/core/db_schema.py` 会为旧 `users` 表补充 `original_username` 和 `username_change_remaining` 字段。

```python
# schemas/user.py
from pydantic import BaseModel, EmailStr, Field, field_validator
from typing import Optional
import re

class UserUpdateRequest(BaseModel):
    nickname: Optional[str] = Field(None, max_length=10)
    email: Optional[EmailStr] = None
    phone: Optional[str] = Field(None, pattern=r'^1[3-9]\d{9}$')
    email_code: Optional[str] = Field(None, min_length=6, max_length=6)

    @field_validator('nickname')
    @classmethod
    def validate_nickname(cls, v):
        if v and not re.match(r'^[\u4e00-\u9fa5a-zA-Z0-9_]+$', v):
            raise ValueError('昵称只能包含中文、字母、数字和下划线')
        return v
```

#### 服务层

```python
# services/user_service.py
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select, update
from app.models.user import User
from app.utils.validators import validate_unique_email, validate_unique_phone

class UserService:
    async def update_user_info(
        self,
        db: AsyncSession,
        user_id: int,
        update_data: UserUpdateRequest
    ) -> User:
        # 验证邮箱唯一性
        if update_data.email:
            await validate_unique_email(db, update_data.email, exclude_id=user_id)

        # 验证手机号唯一性
        if update_data.phone:
            await validate_unique_phone(db, update_data.phone, exclude_id=user_id)

        # 更新用户信息
        update_values = {}
        if update_data.nickname:
            update_values['nickname'] = update_data.nickname
        if update_data.email:
            update_values['email'] = update_data.email
        if update_data.phone:
            update_values['phone'] = update_data.phone

        if update_values:
            await db.execute(
                update(User)
                .where(User.id == user_id)
                .values(**update_values)
            )
            await db.commit()

        return await self.get_user_by_id(db, user_id)
```

### 10. 代码组织建议

#### Pydantic模型

```python
# schemas/user.py
from pydantic import BaseModel, Field, field_validator
import re

class ChangePasswordRequest(BaseModel):
    old_password: str = Field(..., min_length=8, max_length=20)
    new_password: str = Field(..., min_length=8, max_length=20)
    confirm_password: str = Field(..., min_length=8, max_length=20)

    @field_validator('new_password')
    @classmethod
    def validate_password_strength(cls, v):
        if not re.match(r'^(?=.*[a-zA-Z])(?=.*\d).+$', v):
            raise ValueError('密码必须包含字母和数字')
        return v

    @field_validator('confirm_password')
    @classmethod
    def passwords_match(cls, v, info):
        if 'new_password' in info.data and v != info.data['new_password']:
            raise ValueError('两次输入的密码不一致')
        return v
```

#### 路由函数

```python
# api/v1/users.py
@router.post("/me/change-password", response_model=ApiResponse)
async def change_password(
    request: ChangePasswordRequest,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db)
):
    # 验证原密码
    if not verify_password(request.old_password, current_user.password_hash):
        raise HTTPException(
            status_code=400,
            detail="原密码错误，请重新输入"
        )

    # 验证新密码不能与原密码相同
    if verify_password(request.new_password, current_user.password_hash):
        raise HTTPException(
            status_code=400,
            detail="新密码不能与原密码相同"
        )

    # 更新密码
    new_hash = hash_password(request.new_password)
    await user_service.change_password(db, current_user.id, new_hash)

    return ApiResponse(
        code=200,
        message="密码修改成功，请重新登录"
    )
```

#### Pydantic模型

```python
# schemas/learning.py
from pydantic import BaseModel
from datetime import datetime
from typing import Optional
from enum import Enum

class TimeRangeEnum(str, Enum):
    RECENT_7 = "recent_7"
    RECENT_30 = "recent_30"
    ALL = "all"

class LearningRecordItem(BaseModel):
    course_id: int
    course_title: str
    course_cover: str
    last_section_id: int
    last_section_title: str
    last_learn_at: datetime
    course_status: str

class LearningRecordResponse(BaseModel):
    items: list[LearningRecordItem]
    total: int
    page: int
    page_size: int
    total_pages: int
```

#### 服务层

```python
# services/learning_service.py
from datetime import datetime, timedelta
from sqlalchemy import select, and_

class LearningService:
    async def get_learning_records(
        self,
        db: AsyncSession,
        user_id: int,
        time_range: TimeRangeEnum,
        page: int,
        page_size: int
    ) -> LearningRecordResponse:
        # 计算时间条件
        time_condition = None
        if time_range == TimeRangeEnum.RECENT_7:
            time_condition = datetime.now() - timedelta(days=7)
        elif time_range == TimeRangeEnum.RECENT_30:
            time_condition = datetime.now() - timedelta(days=30)

        # 构建查询
        query = (
            select(
                Course.id.label('course_id'),
                Course.title.label('course_title'),
                Course.cover_url.label('course_cover'),
                Section.id.label('last_section_id'),
                Section.title.label('last_section_title'),
                LearningProgress.last_learn_at,
                Course.status.label('course_status')
            )
            .join(Section, LearningProgress.section_id == Section.id)
            .join(Chapter, Section.chapter_id == Chapter.id)
            .join(Course, Chapter.course_id == Course.id)
            .where(LearningProgress.user_id == user_id)
        )

        if time_condition:
            query = query.where(LearningProgress.last_learn_at >= time_condition)

        query = query.order_by(LearningProgress.last_learn_at.desc())

        # 分页
        offset = (page - 1) * page_size
        query = query.offset(offset).limit(page_size)

        result = await db.execute(query)
        items = result.all()

        # 统计总数
        count_query = (
            select(func.count())
            .select_from(LearningProgress)
            .where(LearningProgress.user_id == user_id)
        )
        total = await db.scalar(count_query)

        return LearningRecordResponse(
            items=items,
            total=total,
            page=page,
            page_size=page_size,
            total_pages=(total + page_size - 1) // page_size
        )
```
