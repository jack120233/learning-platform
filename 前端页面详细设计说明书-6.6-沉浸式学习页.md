## 5. 核心业务逻辑与交互

### 5.1 初始化流程 (onMounted)

```
onMounted
 ├─ 1. 路由参数解析
 │    ├─ courseId = Number(route.params.courseId)
 │    ├─ targetSectionId = Number(route.query.sectionId) || null
 │    └─ targetResourceId = Number(route.query.resourceId) || null
 │
 ├─ 2. 课程数据加载
 │    ├─ 检查 learnStore.currentCourseId === courseId
 │    │   ├─ 匹配 → 直接使用 store 中的章节数据（从详情页传递）
 │    │   └─ 不匹配 → 重新请求
 │    │       └─ GET /api/v1/courses/{courseId} → 提取 chapters 存入 store
 │    │
 │    └─ 校验课程状态
 │        ├─ status !== 'published' → 展示下架/不可访问提示
 │        └─ status === 'published' → 继续
 │
 ├─ 3. 构建侧边栏目录树
 │    └─ chapterTree = buildChapterTree(learnStore.currentCourseChapters)
 │        ├─ 默认展开第一个章节
 │        └─ 为每个 section/resource 初始化进度为 0
 │
 ├─ 4. 确定初始资源
 │    ├─ 优先级 1：路由 query 中指定的 sectionId + resourceId
 │    ├─ 优先级 2：继续学习接口返回的最后资源
 │    │   └─ GET /api/v1/learning/courses/{courseId}/continue
 │    └─ 优先级 3：课程第一个章节 → 第一个小节 → 第一个资源
 │
 ├─ 5. 加载初始资源
 │    └─ learnStore.switchResource(sectionId, resourceId)
 │        ├─ GET /api/v1/learning/resources/{resourceId}/play → 获取播放地址
 │        ├─ GET /api/v1/learning/progress?section_id&resource_id → 获取历史进度
 │        └─ 更新 activeResource 状态
 │
 ├─ 6. 启动进度上报定时器
 │    └─ progressSyncTimer = setInterval(syncProgress, 30_000)
 │
 ├─ 7. 注册网络状态监听
 │    ├─ window.addEventListener('online', handleOnline)
 │    └─ window.addEventListener('offline', handleOffline)
 │
 └─ 8. 注册页面离开事件
      ├─ window.addEventListener('beforeunload', handleBeforeUnload)
      └─ onBeforeRouteLeave → 保存当前进度 → cleanup
```

### 5.2 ★ 进度上报防抖策略（重点设计）

#### 5.2.1 设计目标

- 每 30 秒自动保存一次播放进度
- 暂停/切换资源/离开页面时**立即**保存
- 避免高频 `timeupdate` 事件导致过多 API 调用
- 网络中断时缓存进度，恢复后重新同步
- 快速拖动进度条时不触发大量请求

#### 5.2.2 三层防抖架构

```
┌────────────────────────────────────────────────────────────┐
│  Layer 1: 播放器事件层（高频）                               │
│  ┌─────────────────────────────────────────────────────┐   │
│  │ timeupdate 事件（约 250ms 一次，4次/秒）              │   │
│  │  → 仅更新内存中的 currentTime（零 API 调用）          │   │
│  │  → updatePlayProgress(currentTime, totalTime)        │   │
│  └─────────────────────────────────────────────────────┘   │
│                          ↓                                 │
│  Layer 2: 节流采样层（中频）                                │
│  ┌─────────────────────────────────────────────────────┐   │
│  │ 30 秒定时器 setInterval（固定频率）                   │   │
│  │  → 读取内存中最新的 currentTime                       │   │
│  │  → 对比上次上报位置 lastReportedTime                  │   │
│  │  → 变化 > 5 秒才实际发送请求（避免暂停时重复上报）      │   │
│  │  → syncProgressToServer()                            │   │
│  └─────────────────────────────────────────────────────┘   │
│                          ↓                                 │
│  Layer 3: 立即保存层（低频，关键时刻）                       │
│  ┌─────────────────────────────────────────────────────┐   │
│  │ 触发条件：                                            │   │
│  │  • 用户点击暂停                                       │   │
│  │  • 用户切换资源（switchResource）                      │   │
│  │  • 用户离开页面（beforeunload / onBeforeRouteLeave）  │   │
│  │  • 播放完成（ended 事件）                              │   │
│  │  • 网络恢复（online 事件）                             │   │
│  │ 行为：                                                │   │
│  │  → 立即调用 syncProgressToServer()                    │   │
│  │  → 使用 navigator.sendBeacon 作为 beforeunload 降级   │   │
│  └─────────────────────────────────────────────────────┘   │
└────────────────────────────────────────────────────────────┘
```

#### 5.2.3 核心实现代码设计

```typescript
// composables/useProgressSync.ts

interface UseProgressSyncOptions {
  intervalMs: number           // 定时上报间隔，默认 30000
  minDeltaSeconds: number      // 最小变化阈值，默认 5
}

function useProgressSync(options: UseProgressSyncOptions = { intervalMs: 30000, minDeltaSeconds: 5 }) {
  const learnStore = useLearnStore()
  let syncTimer: ReturnType<typeof setInterval> | null = null
  let lastReportedTime = 0
  let isSyncing = false

  /**
   * Layer 1: 播放器 timeupdate 回调（高频，仅更新内存）
   * 由播放器组件在 @timeupdate 中调用
   */
  function onTimeUpdate(currentTime: number, totalTime: number): void {
    learnStore.updatePlayProgress(currentTime, totalTime)
    // 更新侧边栏进度百分比（内存操作，无 API 调用）
  }

  /**
   * Layer 2: 定时上报（30s 一次）
   */
  function startPeriodicSync(): void {
    syncTimer = setInterval(async () => {
      const active = learnStore.activeResource
      if (!active || active.playState !== 'playing') return

      const delta = Math.abs(active.currentTime - lastReportedTime)
      if (delta < options.minDeltaSeconds) return  // 变化不足，跳过

      await doSync(false)
    }, options.intervalMs)
  }

  /**
   * Layer 3: 立即保存
   */
  async function immediateSync(): Promise<void> {
    await doSync(true)
  }

  /**
   * 实际同步逻辑（防并发锁）
   */
  async function doSync(force: boolean): Promise<void> {
    if (isSyncing && !force) return  // 非强制模式下，正在同步则跳过
    const active = learnStore.activeResource
    if (!active || !active.resourceId) return

    isSyncing = true
    try {
      const request: SaveProgressRequest = {
        section_id: active.sectionId,
        resource_id: active.resourceId,
        current_time: Math.floor(active.currentTime),
        total_time: Math.floor(active.totalTime),
        is_completed: active.isCompleted
      }

      await learningApi.saveProgress(request)
      lastReportedTime = active.currentTime

      // 更新本地进度缓存
      learnStore.progressCache.set(active.resourceId, {
        currentTime: active.currentTime,
        totalTime: active.totalTime,
        isCompleted: active.isCompleted,
        lastSyncedAt: Date.now()
      })
    } catch (error) {
      // 静默失败，不打断用户学习
      // 如果是网络错误，加入离线队列
      if (!navigator.onLine) {
        offlineQueue.push({
          section_id: active.sectionId,
          resource_id: active.resourceId,
          current_time: Math.floor(active.currentTime),
          total_time: Math.floor(active.totalTime),
          is_completed: active.isCompleted
        })
      }
      console.warn('[ProgressSync] 进度保存失败，将在下次重试', error)
    } finally {
      isSyncing = false
    }
  }

  /**
   * beforeunload 降级：使用 sendBeacon 确保页面关闭时数据不丢失
   */
  function onBeforeUnload(): void {
    const active = learnStore.activeResource
    if (!active || !active.resourceId) return

    const payload = JSON.stringify({
      section_id: active.sectionId,
      resource_id: active.resourceId,
      current_time: Math.floor(active.currentTime),
      total_time: Math.floor(active.totalTime),
      is_completed: active.isCompleted
    })

    // sendBeacon 是异步的，不会阻塞页面关闭
    navigator.sendBeacon(
      '/api/v1/learning/progress',
      new Blob([payload], { type: 'application/json' })
    )
  }

  /** 网络恢复：刷新离线队列 */
  async function flushOfflineQueue(): Promise<void> {
    while (offlineQueue.length > 0) {
      const request = offlineQueue.shift()!
      try {
        await learningApi.saveProgress(request)
      } catch {
        offlineQueue.unshift(request)  // 失败则放回队列
        break
      }
    }
  }

  function stopPeriodicSync(): void {
    if (syncTimer) {
      clearInterval(syncTimer)
      syncTimer = null
    }
  }

  onUnmounted(() => {
    stopPeriodicSync()
  })

  return {
    onTimeUpdate,
    startPeriodicSync,
    stopPeriodicSync,
    immediateSync,
    onBeforeUnload,
    flushOfflineQueue
  }
}
```

#### 5.3.2 防竞态控制

```typescript
// composables/useResourceSwitch.ts

let switchAbortController: AbortController | null = null
let currentSwitchId = 0  // 自增 ID，防止旧请求覆盖新结果

async function switchResource(sectionId: number, resourceId: number): Promise<void> {
  // 1. 取消上一次未完成的切换
  if (switchAbortController) {
    switchAbortController.abort()
  }
  switchAbortController = new AbortController()
  const thisSwitchId = ++currentSwitchId

  // 2. 保存当前资源进度
  await progressSync.immediateSync()

  // 3. 重置状态
  learnStore.activeResource = {
    resourceId,
    sectionId,
    resourceType: 'video',  // 临时值，加载后更新
    loadState: 'loading',
    playState: 'stopped',
    currentTime: 0,
    totalTime: 0,
    // ... 其他字段初始化
  } as ActiveResourceState

  try {
    // 4. 并行加载
    const [playInfoRes, progressRes] = await Promise.allSettled([
      learningApi.getResourcePlayUrl(resourceId, { signal: switchAbortController.signal }),
      learningApi.getProgress(sectionId, resourceId, { signal: switchAbortController.signal })
    ])

    // 5. 竞态校验
    if (currentSwitchId !== thisSwitchId) {
      return  // 已被新的切换覆盖，丢弃结果
    }

    // 6. 处理播放地址
    if (playInfoRes.status === 'fulfilled') {
      const playInfo = playInfoRes.value.data
      learnStore.activeResource!.fileUrl = playInfo.file_url
      learnStore.activeResource!.resourceType = playInfo.resource_type
      learnStore.activeResource!.totalTime = playInfo.duration || 0
      learnStore.activeResource!.loadState = 'ready'
    } else {
      learnStore.activeResource!.loadState = 'error'
      learnStore.activeResource!.errorMessage = '资源加载失败'
      return
    }

    // 7. 恢复进度
    if (progressRes.status === 'fulfilled' && progressRes.value.data) {
      const progress = progressRes.value.data
      learnStore.activeResource!.currentTime = progress.current_time
      learnStore.activeResource!.isCompleted = progress.is_completed
    }

  } catch (error) {
    if ((error as Error).name === 'AbortError') return  // 被取消，忽略
    learnStore.activeResource!.loadState = 'error'
    learnStore.activeResource!.errorMessage = '网络异常，请检查网络后重试'
  }
}
```

#### getNextResource 遍历算法

```typescript
function getNextResource(): { sectionId: number; resourceId: number } | null {
  const chapters = learnStore.currentCourseChapters
  const active = learnStore.activeResource
  if (!active) return null

  let found = false

  for (const chapter of chapters) {
    for (const section of chapter.sections) {
      for (const resource of section.resources) {
        if (found) {
          // 找到当前资源之后的第一个资源
          return { sectionId: section.section_id, resourceId: resource.resource_id }
        }
        if (resource.resource_id === active.resourceId) {
          found = true  // 标记找到当前资源
        }
      }
    }
  }

  return null  // 已是最后一个资源
}

// composables/useKeyboardShortcuts.ts
function useKeyboardShortcuts(videoRef: Ref<HTMLVideoElement | null>) {
  function handleKeydown(e: KeyboardEvent): void {
    // 忽略输入框中的按键
    if (['INPUT', 'TEXTAREA', 'SELECT'].includes((e.target as HTMLElement).tagName)) return

    const video = videoRef.value
    if (!video) return

    switch (e.code) {
      case 'Space':
        e.preventDefault()
        video.paused ? video.play() : video.pause()
        break
      case 'ArrowLeft':
        e.preventDefault()
        video.currentTime = Math.max(0, video.currentTime - 5)
        break
      case 'ArrowRight':
        e.preventDefault()
        video.currentTime = Math.min(video.duration, video.currentTime + 5)
        break
      case 'KeyF':
        e.preventDefault()
        document.fullscreenElement ? document.exitFullscreen() : video.requestFullscreen()
        break
      case 'KeyM':
        e.preventDefault()
        video.muted = !video.muted
        break
    }
  }

  onMounted(() => document.addEventListener('keydown', handleKeydown))
  onUnmounted(() => document.removeEventListener('keydown', handleKeydown))
}

  // 1. 立即保存当前进度

  // 2. 停止定时器
  progressSync.stopPeriodicSync()

  // 3. 清理播放器资源
  learnStore.cleanup()

  // 4. 移除事件监听
  window.removeEventListener('beforeunload', progressSync.onBeforeUnload)
  window.removeEventListener('online', handleOnline)
  window.removeEventListener('offline', handleOffline)

  next()
})
```
